model AuditLog {
  id        String      @id @default(cuid())
  action    AuditAction
  actorId   String
  targetId  String?
  createdAt DateTime    @default(now())

  actor User @relation(fields: [actorId], references: [id])

  @@index([actorId, createdAt])
  @@index([createdAt])
}
