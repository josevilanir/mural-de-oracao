   ```ts
     dsn: process.env.SENTRY_DSN,
     tracesSampleRate: 0.1,
     environment: process.env.NODE_ENV,
     enabled: !!process.env.SENTRY_DSN,
   });
   ```

3. Create `sentry.client.config.ts` at project root:
   ```ts

   Sentry.init({
     dsn: process.env.NEXT_PUBLIC_SENTRY_DSN,
     tracesSampleRate: 0.1,
     environment: process.env.NODE_ENV,
     enabled: !!process.env.NEXT_PUBLIC_SENTRY_DSN,
     replaysSessionSampleRate: 0,
     replaysOnErrorSampleRate: 0,
   });
   ```

4. Create `sentry.edge.config.ts` at project root:
   ```ts
   import * as Sentry from "@sentry/nextjs";

   Sentry.init({
     dsn: process.env.SENTRY_DSN,
     tracesSampleRate: 0.1,
     environment: process.env.NODE_ENV,
     enabled: !!process.env.SENTRY_DSN,
   });
   ```

5. Create `instrumentation.ts` at project root:
   ```ts
   export async function register() {
     if (process.env.NEXT_RUNTIME === "nodejs") {
       await import("./sentry.server.config");
     }
     if (process.env.NEXT_RUNTIME === "edge") {
       await import("./sentry.edge.config");
     }
   }

   export const onRequestError = (await import("@sentry/nextjs")).sentryOnRequestError;
   ```
   NOTE: The `onRequestError` export is key — it is the Next.js 14 instrumentation hook that automatically captures errors from server actions and API routes without any manual `captureException` calls.

   ```
   NEXT_PUBLIC_SENTRY_DSN="https://examplePublicKey@o0.ingest.sentry.io/0"
   ```
