    **Step 1 — Modify middleware.ts:**
    Add a check for mutation methods. If the method is not 'GET' (or 'HEAD'/'OPTIONS'), verify that the 'Origin' header matches the 'Host' header.

    ```typescript
    const origin = req.headers.get('origin');
    const host = req.headers.get('host');
    const forwardedHost = req.headers.get('x-forwarded-host');
    const expectedHost = forwardedHost || host;

    // Exempt NextAuth routes
    if (req.nextUrl.pathname.startsWith('/api/auth')) {
      return response;
    }

    if (['POST', 'PUT', 'PATCH', 'DELETE'].includes(req.method)) {
      if (origin) {
        const originHost = new URL(origin).host;
        if (originHost !== expectedHost) {
          return new NextResponse('Invalid Origin', { status: 403 });
        }
      }
    }
    ```

    Note: Be careful with the exact implementation to ensure it doesn't break legitimate flows.
